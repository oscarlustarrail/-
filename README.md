#include <iostream>
using namespace std;
int main(){
	int a = 234;
	int b;
	cout <<"猜數字，1~1000"<<endl; 
	while(true){
		cin >> b;
		if(b>1000||b<1){
			cout << "輸入1~1000"<< endl;
			continue;
		} 
	if(b>a){
		cout << "太大"<< endl;
	}else if(b<a){
		cout << "太小"<< endl;
	} else{
		cout << "恭喜"<< endl;
	}
}
}
